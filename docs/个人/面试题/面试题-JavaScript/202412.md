### 2、省略操作符的全局变量

- 使用var操作符定义的变量将成为该变量的`作用域中的局部变量`，`函数退出变量销毁`

- 省略var操作符会创建一个全局变量

```javascript
(function() {
      var a = b = 5; //等同于 var a = b, b = 5
  })();   
console.log(b); //5
console.log(a); //Uncaught ReferenceError: a is not defined
```

### 6、typeof 和 instanceof

typeof一般用于检测基本数据类型，返回对应数据类型

```javascript
console.log(typeof undefined); // "undefined"
console.log(typeof true);      // "boolean"
console.log(typeof 42);        // "number"
console.log(typeof "hello");   // "string"
console.log(typeof {});        // "object"
console.log(typeof []);        // "object"
console.log(typeof null);      // "object" (特殊情况)
console.log(typeof function(){}); // "function"
console.log(typeof Symbol());  // "symbol"
console.log(typeof 10n);       // "bigint"

```

instanceof检测对象是否为某种复杂引用类型，返回boolean

```javascript
console.log({} instanceof Object);           // true
console.log([] instanceof Array);            // true
console.log(function(){} instanceof Function); // true
console.log(new Date() instanceof Date);     // true

function MyClass() {}
let myInstance = new MyClass();
console.log(myInstance instanceof MyClass);  // true
```

### 10、尾调用

`一个函数的调用 发生 在另一个函数的尾部`。该种方式可使用新函数复用原函数的栈帧，避免栈溢出的风险，如斐波那契递归调用。

```javascript
// 非尾递归
function factorial(n) {
  if (n === 1) return 1;
  return n * factorial(n - 1);
}

// 尾递归,返回的只是递归调用
function factorialTail(n, acc = 1) {
  if (n === 1) return acc;
  return factorialTail(n - 1, n * acc);
}

```
