### 1.1 原始数据类型


### 1.2 其他类型

#### any

- 表示可以赋予任何类型, 可以使用任何类型的属性和方法

```typescript
let anyThing: any = "Tom";
```



#### unknown

- 表示不知道类型的变量，需要使用类型断言确定类型

```typescript
function isUser(obj: unknown): obj is { name: string; age: number } {
  return (
    typeof obj === 'object' &&
    obj !== null &&
    'name' in obj &&
    'age' in obj &&
    typeof (obj as any).name === 'string' &&
    typeof (obj as any).age === 'number'
  );
}
```



#### never

- never 是任何值的子类型、表示永远不存在值的类型
- 确保开发时候想到所有情况，用于穷尽检查(switch 或 ifelse)、永不返回（死循环 抛错函数）

```typescript
// ✅ 有 never 检查 - 编译时报错
type Status = 'loading' | 'success' | 'error';

function handleStatus(status: Status) {
  switch (status) {
    case 'loading': console.log('加载中'); break;
    case 'success': console.log('成功'); break;
    // case 'error': console.log('错误'); break;  //取消这行注释就能编译通过
  }
  const _: never = status;  // ❌ 编译错误！
  // 错误：Type '"error"' is not assignable to type 'never'， 提示你忘记处理 'error' 情况
}
```

:bulb: never 只是用于`编译时检查问题`，确保各个情况都考虑到，启动后不受影响

:bulb: `TypeScript 能推导出代码执行路径`，知道哪些分支可能执行 , 开发时的安全网



#### void

- 返回了，只是返回undefined

```typescript
function warnUser(): void {
  console.log("This is my warning message");
}
```



### 1.3 接口

#### 定义接口




#### 可选属性

```typescript
interface Person {
```

#### 任意数量属性

- 确定属性和可选属性必须为任意属性的子类型

```typescript
interface RequestParams {
  method: 'GET' | 'POST';
  url: string;
  [param: string]: string | number;  //允许对象拥有任意数量的额外属性，只要属性名是 string类型，属性值是 string | number类型
}
```



#### 只读属性

```typescript
interface Person {
  readonly id: number;
  name: string;
  age?: number;
  [propName: string]: string | number;
}
```



#### 接口继承(重写、多继承)

```typescript
interface Person {
  name: string;
  age: number;
  sayHello(): void; 
}

interface StudentAbility {
  study(): void; 
  getScore(): number;
}

interface SuperStudent extends Person, StudentAbility {
  // 继承了：name, age, sayHello, study, getScore

  // 重写（override）父类方法
  sayHello(): void; 

  // 自己新增属性 + 方法
  major: string;
  doHomework(): void;
}
```

#### 函数重载

- 同一个函数接受不同数量或类型的参数

```typescript
function reverse(x: number): number;
function reverse(x: string): string;
function reverse(x: number | string): number | string | void {
  if (typeof x === "number") {
    return Number(x.toString().split("").reverse().join(""));
  } else if (typeof x === "string") {
    return x.split("").reverse().join("");
  }
}
```



#### 实现箭头函数重载

- 使用 interface 和 as 实现箭头函数重载;

```typescript
interface FN {
  (x: number): number;
  (x: number, y: number): number;
}
const reverse1: FN = (x: number, y?: number): number => {
  if (y) {
    return x + y;
  } else {
    return x;
  }
};

const reverse2 = ((x: number, y?: number): number => {
  if (y) {
    return x + y;
  } else {
    return x;
  }
}) as FN;  // 用于确保类型 告诉TS编译器放宽检查
```



### 1.5 联合类型和交叉类型

#### 联合类型

- 联合类型表示几个类型中的一个;
- 只能访问联合类型共有的方法和属性;

```typescript
let myFavoriteNumber: string | number;
myFavoriteNumber = "seven";
myFavoriteNumber = 7;
```



#### 交叉类型

- 交叉类型将多个类型合并在一起;
- 具有多个类型的属性和方法;

```typescript
interface ErrorHandling {
  success: boolean;
  error?: { message: string };
}

interface ArtworksData {
  artworks: { title: string }[];  //对象数组，对象必须有title属性，且是string
}
type ArtworksResponse = ArtworksData & ErrorHandling;
```

### 二、 泛型

#### 2.1 :fire: 基本概念

- 定义函数, 接口或类时;
- 不确定具体类型;
- 在使用时确定;



```typescript
// 普通函数
function createArray<T>(length: number, value: T): Array<T> {
  let result: T[] = [];
  for (let i = 0; i < length; i++) {
    result[i] = value;
  }
  return result;
}
// 箭头函数代默认类型
const createArray = <T = string>(length: number, value: T): Array<T> => {
  let result: T[] = [];
  for (let i = 0; i < length; i++) {
    result[i] = value;
  }
  return result;
};
// 确定类型
createArray<string>(3, "x"); // ['x', 'x', 'x']
```



#### 2.2 泛型接口

```typescript
interface CreateArrayFunc {
  <T>(length: number, value: T): Array<T>;
}

const createArray: CreateArrayFunc = <T>(
  length: number,
  value: T
): Array<T> => {
  let result: T[] = [];
  for (let i = 0; i < length; i++) {
    result[i] = value;
  }
  return result;
};
```



#### 2.3 :fire: 泛型类

```typescript
class GenericNumber<T> {
  zeroValue: T;
  add: (x: T, y: T) => T;
}
let myGenericNumber = new GenericNumber<number>();
myGenericNumber.zeroValue = 0;
myGenericNumber.add = function (x, y) {
  return x + y;
};
```
