# Object

## 一、基础

### 1.1 创建对象

- new 操作符;
- object literal notation (推荐使用);
- Object.create()

```javascript
let person = new Object();
let person = {};
const person2 = Object.create(person);
```



### 1.2 设置属性

```javascript
let person = new Object();
person.name = "Nicholas";
people["name"] = "Nicholas";
let person = {
  name: "Nicholas";
};
```



## 二、语法

### 2.1 对象属性

- 表示属性值行为, 必须通过 defineProperty() 自定义;
- 属性行为;
  - configurable: 表示属性可被删除/重新定义, 默认为 true;
  - enumerable: 表示属性可枚举, 应用于 for-in 循环, 默认为 true;
  - writable: 表示属性值可被改变, 默认为 true;
  - value: 表示属性值, 默认为 undefined;
  - get: 当属性被读时调用的函数, 默认值为 undefined;
  - set: 当属性被写时调用的函数, 默认值为 undefined;
    - 只定义 get, 表明只读;
    - 只定义 set, 表示只写;



### 2.2 对象语法增强

属性值简写

```javascript
// 传统
let name = "Matt";
let person = {
  name: name,
};

// 属性值简写
let name = "Matt";
let person = {
  name,
};
```



可计算属性

```javascript
// 传统
const nameKey = "name";
let person = {};
person[nameKey] = "Matt"; // { name: 'Matt' }

// 可计算属性
const jobKey = 333;
let person1 = {
  [jobKey + 2]: "Matt",
}; // { name: 'Matt' }
console.log(person1); // { 335: 'Matt' }
```



简写方法名

```javascript
// 传统
let person = {
  sayName: function (name) {
    console.log("My name is ${name}");
  },
};

// 简写方法名, 广泛应用于 class
let person = {
  sayName(name) {
    console.log("My name is ${name}");
  },
};
```



### 2.3 对象解构

object

```javascript
// 未赋值的属性为 undefined
let person = {
  name: "Matt",
  age: 27,
};
let { name, job } = person;
console.log(name); // Matt
console.log(job); // undefined

// 函数参数使用解构
let person = {
  name: "Matt",
  age: 27,
};
function printPerson(foo, { name: personName, age: personAge }, bar) {
  console.log(arguments);
  console.log(personName, personAge);
}
printPerson("1st", person, "2nd");
// ['1st', { name: 'Matt', age: 27 }, '2nd']
// 'Matt', 27
```



数组解构

```javascript
// 基本解构
let colors = ["red", "green", "blue"];
let [firstColor, secondColor] = colors;
console.log(firstColor);  // "red"
console.log(secondColor); // "green"

// 跳过元素
let [, , thirdColor] = colors;
console.log(thirdColor);  // "blue"

// 解构剩余元素
let [primary, ...restColors] = colors;
console.log(primary);     // "red"
console.log(restColors);  // ["green", "blue"]
```



字符串解构

```javascript
// 字符串解构
let str = "hello";
let [c1, c2, c3] = str;
console.log(c1, c2, c3);  // "h", "e", "l"
console.log(firstChar, restChars);  // "你", ["好", "世", "界"]
```



## 三、设计模式

### 3.1 工厂模式

```javascript
// 无法自定义返回对象类型
function createPerson(name, age, job) {
  let o = new Object();
  o.name = name;
  o.age = age;
  o.job = job;
  o.sayName = function () {
    console.log(this.name);
  };
  return o;
}
let person1 = createPerson("Nicholas", 29, "Software Engineer");
```



### 3.2 构造函数模型

```javascript
function Person(name, age, job) {
  this.name = name;
  this.age = age;
  this.job = job;
  this.sayName = function () {
    console.log(this.name);
  };
}

let person1 = new Person("Nicholas", 29, "Software Engineer");
let person2 = new Person("Greg", 27, "Doctor");
console.log(person1.constructor === Person); // true
console.log(person2.constructor === Person); // true
// Person 实例可被表示为自定义类型
console.log(person1 instanceof Person); // true
```



| 特性         | 构造函数模式                              | 工厂函数模式                              |
| ------------ | ----------------------------------------- | ----------------------------------------- |
| **创建方式** | `new Person(...)`                         | `createPerson(...)`                       |
| **内部实现** | 使用 `this`绑定属性                       | 手动创建对象并返回                        |
| **原型链**   | 实例的 `__proto__`指向 `Person.prototype` | 实例的 `__proto__`指向 `Object.prototype` |

## 四、继承

### 4.1 盗用构造函数（缺失方法）

```javascript
function SuperType() {
  this.colors = ["red", "blue", "green"];
}
function SubType() {
  // 继承 SuperType 属性
  SuperType.call(this);
  this.age = 29;
}
let instance1 = new SubType();
instance1.colors.push("black");
console.log(instance1.colors); // "red,blue,green,black"
console.log(instance1.age); // 29

let instance2 = new SubType();
console.log(instance2.colors); // "red,blue,green"
```

:warning: 缺点

- SuperType() 中的方法无法复用;
- 无法使用 SuperType.Prototype 中的方法;



### 4.2 组合继承（属性重复）

```javascript
// 原型链和盗用构造函数的结合
// 使用 prototype chaining 继承 prototype 上的属性
// 使用 Constructor Stealing 继承 constructor function 上的属性
function SuperType(name) {
  this.name = name;
  this.colors = ["red", "blue", "green"];
}
SuperType.prototype.sayName = function () {
  console.log(this.name);
};
function SubType(name, age) {
  // 继承 SuperType 属性
  SuperType.call(this, name);
  this.age = age;
}
// 继承 SuperType 方法
SubType.prototype = new SuperType();
SubType.prototype.sayAge = function () {
  console.log(this.age);
};
```

![组合继承](./%E5%AF%B9%E8%B1%A1.assets/2023-11-21-18-34-28-e3329c98271e882f40111746fa9f9916.png)

:warning: 缺点

- 调用两次 SuperType();
- SubType 的 prototype 具有冗余属性; （多次声明实例的name 和 color） 



### 4.3 寄生式组合继承

```javascript
function Sup(name) {
  this.name = name;
}

Sup.prototype.sayName = function () {
  console.log(this.name);
};

function Sub(name, age) {
  Sup.call(this, name);  // 1. 继承实例属性
  this.age = age;
}

// 2. 继承原型方法（避免调用父类构造函数）
Sub.prototype = Object.create(Sup.prototype);

// 3. 修正 constructor 指向
Sub.prototype.constructor = Sub;

// 4. 添加子类特有的原型方法
Sub.prototype.sayAge = function () {
  console.log(this.age);
};
```



### 4.4 class

```javascript
class Sup {
  constructor(name) {
    this.name = name;
  }
  
  sayName() {
    console.log(this.name);
  }
}

class Sub extends Sup {
  constructor(name, age) {
    super(name);
    this.age = age;
  }
  
  sayAge() {
    console.log(this.age);
  }
}
```



## 五、最佳实践

### 5.1 不可修改

- 使用defineProperty定义不可修改，如果可以修改使用字面量对象操作
- 已经创建的使用Object.freeze(obj)



### 5.2 数据劫持

- 使用Proxy 创建代理obj
- 拦截操作通过Reflect统一管理

```javascript
const obj = {
  name: "刘逍",
  age: 20,
};

const p = new Proxy(obj, {
  get(target, propName) {
    console.log(`${propName}属性被读取`);
    return Reflect.get(target, propName);
  },

  set(target, propName, value) {
    console.log(`${propName}属性被修改`);
    Reflect.set(target, propName, value);
  },

  deleteProperty(target, propName) {
    console.log(`${propName}属性被删除`);
    return Reflect.deleteProperty(target, propName);
  },
});
```
