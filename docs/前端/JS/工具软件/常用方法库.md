## clickOut

```
export default {
  mounted(el, binding) {
    const onClickOutside = (event) => {
      if (!(el === event.target || el.contains(event.target))) {
        binding.value(event)
      }
    }
    
    document.addEventListener('click', onClickOutside)
    el._clickOutside = onClickOutside
  },
  unmounted(el) {
    document.removeEventListener('click', el._clickOutside)
    delete el._clickOutside
  }
}

挂载：
app.directive('click-outside', clickOutside)
```

## query

```javascript
function objToQuery(obj) {
  if (!obj) return ''
  return Object.entries(obj)
    .filter(([, val]) => val !== undefined && val !== null)
    .map(([key, val]) => `${key}=${val}`)
    .join('&')
}

function getComponentData(payload){
  let url = `/edu-plan-v2/achievementRate/getAchievementRateModuleData?${objToQuery(payload)}`
  return new Promise((resolve, reject) => {
    httpGet(url)
      .then((res) => {
        resolve(res)
      })
      .catch((err) => {
        reject(err)
      })
  })
}
```

## 深拷贝

```javascript
//深拷贝
/**
 * 深拷贝函数（支持常见数据类型 + 循环引用）
 * @param {any} target - 要拷贝的目标值
 * @param {WeakMap} [cache=new WeakMap()] - 缓存已拷贝对象，解决循环引用（外部无需传递）
 * @returns {any} 拷贝后的新值
 */
function deepClone(target, cache = new WeakMap()) {
  // 1. 基本类型直接返回（值类型无需拷贝）
  if (target === null || typeof target !== 'object') {
    return target
  }

  // 2. 处理循环引用（已拷贝过的对象直接返回缓存）
  if (cache.has(target)) {
    return cache.get(target)
  }

  let cloneResult
  const targetConstructor = target.constructor

  // 3. 特殊对象类型处理
  switch (targetConstructor) {
    // 3.1 日期对象
    case Date:
      cloneResult = new Date(target)
      break

    // 3.2 正则对象
    case RegExp:
      cloneResult = new RegExp(target.source, target.flags)
      cloneResult.lastIndex = target.lastIndex // 保留正则的lastIndex属性
      break

    // 3.3 Map对象
    case Map:
      cloneResult = new Map()
      cache.set(target, cloneResult) // 先缓存，避免循环引用
      target.forEach((value, key) => {
        cloneResult.set(deepClone(key, cache), deepClone(value, cache))
      })
      break

    // 3.4 Set对象
    case Set:
      cloneResult = new Set()
      cache.set(target, cloneResult) // 先缓存，避免循环引用
      target.forEach(value => {
        cloneResult.add(deepClone(value, cache))
      })
      break

    // 3.5 数组（构造函数是Array）
    case Array:
      cloneResult = []
      cache.set(target, cloneResult)
      target.forEach((item, index) => {
        cloneResult[index] = deepClone(item, cache)
      })
      break

    // 3.6 普通对象（包括自定义对象，排除上述特殊类型）
    default:
      cloneResult = new targetConstructor() // 保持原对象的构造函数（如自定义类实例）
      cache.set(target, cloneResult)
      // 拷贝对象的所有自身属性（包括不可枚举属性）
      Reflect.ownKeys(target).forEach(key => {
        cloneResult[key] = deepClone(target[key], cache)
      })
      break
  }

  return cloneResult
}
```
